import streamlit as st
import pandas as pd
import datetime
import os

# Konfigurasi Halaman
st.set_page_config(page_title="Sistem Gudang APHP", page_icon="📦", layout="wide")

# Fungsi untuk memuat atau membuat database CSV
def load_data(filename, columns):
    if not os.path.exists(filename):
        df = pd.DataFrame(columns=columns)
        df.to_csv(filename, index=False)
    return pd.read_csv(filename)

# File penyimpanan
FILE_STOK = "stok_gudang.csv"
FILE_KONDISI = "kondisi_gudang.csv"

# Load Data
df_stok = load_data(FILE_STOK, ["Tanggal_Masuk", "Nama_Barang", "Kategori", "Jumlah", "Satuan", "Tanggal_Kadaluarsa", "Lokasi_Simpan"])
df_kondisi = load_data(FILE_KONDISI, ["Tanggal", "Waktu", "Suhu_Celcius", "Kelembaban_Persen", "Petugas"])

st.title("📦 Aplikasi Keamanan Pangan & Penggudangan")
st.markdown("**Praktikum Siswa Kelas XI - Agribisnis Pengolahan Hasil Pertanian (APHP)**")
st.divider()

# Sidebar untuk Navigasi Menu
menu = st.sidebar.selectbox("Pilih Menu:", 
    ["Dashboard & Stok", "Catat Barang Masuk", "Catat Barang Keluar", "Log Suhu & Kelembaban"]
)

# 1. MENU DASHBOARD
if menu == "Dashboard & Stok":
    st.header("📊 Dashboard Stok Gudang")
    
    if not df_stok.empty:
        # Peringatan Kadaluarsa (Sistem FEFO)
        st.subheader("⚠️ Peringatan Keamanan Pangan (FEFO)")
        df_stok['Tanggal_Kadaluarsa'] = pd.to_datetime(df_stok['Tanggal_Kadaluarsa'])
        hari_ini = pd.to_datetime(datetime.date.today())
        
        df_stok['Sisa_Hari'] = (df_stok['Tanggal_Kadaluarsa'] - hari_ini).dt.days
        
        barang_kritis = df_stok[df_stok['Sisa_Hari'] <= 30]
        if not barang_kritis.empty:
            for index, row in barang_kritis.iterrows():
                if row['Sisa_Hari'] < 0:
                    st.error(f"KADALUARSA: {row['Nama_Barang']} (Lewat {abs(row['Sisa_Hari'])} hari!) - Segera musnahkan!")
                else:
                    st.warning(f"HAMPIR KADALUARSA: {row['Nama_Barang']} (Sisa {row['Sisa_Hari']} hari) - Gunakan segera!")
        else:
            st.success("Semua barang dalam kondisi aman dari kadaluarsa dekat.")

        st.subheader("Data Stok Saat Ini")
        st.dataframe(df_stok.drop(columns=['Sisa_Hari']), use_container_width=True)
        
        # Tombol download untuk rekap siswa
        csv = df_stok.to_csv(index=False).encode('utf-8')
        st.download_button("Unduh Laporan Stok (CSV)", data=csv, file_name="Laporan_Stok_APHP.csv", mime="text/csv")
    else:
        st.info("Gudang saat ini kosong. Silakan catat barang masuk.")

# 2. MENU BARANG MASUK
elif menu == "Catat Barang Masuk":
    st.header("📥 Form Penerimaan Barang")
    st.write("Catat bahan baku atau produk olahan yang masuk ke gudang.")
    
    with st.form("form_masuk"):
        col1, col2 = st.columns(2)
        with col1:
            tgl_masuk = st.date_input("Tanggal Masuk")
            nama_barang = st.text_input("Nama Barang (Contoh: Tepung Terigu, Minyak Atsiri)")
            kategori = st.selectbox("Kategori", ["Bahan Baku Segar", "Bahan Kering", "Bahan Kimia/BTM", "Produk Jadi", "Kemasan"])
        with col2:
            jumlah = st.number_input("Jumlah", min_value=0.0, step=0.1)
            satuan = st.selectbox("Satuan", ["Kg", "Liter", "Gram", "Pcs", "Botol"])
            tgl_expired = st.date_input("Tanggal Kadaluarsa (Expired Date)")
            lokasi = st.selectbox("Lokasi Penyimpanan", ["Suhu Ruang (Rak Kering)", "Chiller (Pendingin)", "Freezer (Pembeku)"])
        
        submit = st.form_submit_button("Simpan Data Barang Masuk")
        
        if submit:
            if nama_barang != "" and jumlah > 0:
                data_baru = pd.DataFrame([{
                    "Tanggal_Masuk": tgl_masuk, "Nama_Barang": nama_barang, "Kategori": kategori,
                    "Jumlah": jumlah, "Satuan": satuan, "Tanggal_Kadaluarsa": tgl_expired, "Lokasi_Simpan": lokasi
                }])
                df_stok = pd.concat([df_stok, data_baru], ignore_index=True)
                df_stok.to_csv(FILE_STOK, index=False)
                st.success(f"✅ {nama_barang} berhasil ditambahkan ke gudang!")
            else:
                st.error("Nama barang dan jumlah tidak boleh kosong!")

# 3. MENU BARANG KELUAR
elif menu == "Catat Barang Keluar":
    st.header("📤 Form Pengeluaran Barang")
    st.write("Pilih barang yang akan dikeluarkan untuk praktikum / produksi (Terapkan prinsip FEFO).")
    
    if not df_stok.empty:
        with st.form("form_keluar"):
            pilihan_barang = st.selectbox("Pilih Barang Tersedia", df_stok['Nama_Barang'].unique())
            stok_tersedia = df_stok[df_stok['Nama_Barang'] == pilihan_barang]['Jumlah'].sum()
            
            st.info(f"Stok tersedia untuk {pilihan_barang}: {stok_tersedia}")
            jumlah_keluar = st.number_input("Jumlah yang dikeluarkan", min_value=0.0, max_value=float(stok_tersedia), step=0.1)
            
            submit_keluar = st.form_submit_button("Keluarkan Barang")
            
            if submit_keluar and jumlah_keluar > 0:
                # Logika sederhana mengurangi stok berdasarkan index pertama yang ditemukan (simulasi FIFO/FEFO)
                idx = df_stok[df_stok['Nama_Barang'] == pilihan_barang].index[0]
                df_stok.at[idx, 'Jumlah'] -= jumlah_keluar
                
                # Hapus baris jika stok habis
                df_stok = df_stok[df_stok['Jumlah'] > 0]
                df_stok.to_csv(FILE_STOK, index=False)
                
                st.success(f"✅ {jumlah_keluar} {pilihan_barang} berhasil dikeluarkan!")
    else:
        st.warning("Gudang kosong.")

# 4. MENU KONDISI GUDANG (SUHU & KELEMBABAN)
elif menu == "Log Suhu & Kelembaban":
    st.header("🌡️ Form Inspeksi Sanitasi & Kondisi Ruang")
    st.write("Sesuai standar Keamanan Pangan, suhu ruang penyimpanan harus dipantau.")
    
    with st.form("form_suhu"):
        col1, col2 = st.columns(2)
        with col1:
            tgl_cek = st.date_input("Tanggal Pengecekan")
            waktu_cek = st.time_input("Waktu Pengecekan")
            petugas = st.text_input("Nama Petugas / Siswa Piket")
        with col2:
            suhu = st.number_input("Suhu Ruangan (°C)", value=25.0, step=0.5)
            kelembaban = st.number_input("Kelembaban Relatif (%)", value=60.0, step=1.0)
            
        submit_suhu = st.form_submit_button("Catat Log Suhu")
        
        if submit_suhu and petugas != "":
            log_baru = pd.DataFrame([{
                "Tanggal": tgl_cek, "Waktu": waktu_cek, 
                "Suhu_Celcius": suhu, "Kelembaban_Persen": kelembaban, "Petugas": petugas
            }])
            df_kondisi = pd.concat([df_kondisi, log_baru], ignore_index=True)
            df_kondisi.to_csv(FILE_KONDISI, index=False)
            st.success("✅ Log kondisi ruangan berhasil dicatat!")
            
    if not df_kondisi.empty:
        st.subheader("Riwayat Pengecekan Kondisi Gudang")
        st.dataframe(df_kondisi, use_container_width=True)
